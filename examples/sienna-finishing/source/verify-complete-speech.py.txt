"""Transcribe actual complete narration media using the existing offline CPU ASR."""
from pathlib import Path
import argparse,os,json,re,subprocess,time,difflib,hashlib,math

task=Path(__file__).resolve().parents[1]
p=task.parent/'i-cant-hear-anything-on-my/outputs/video-editor-agent/data/projects/nano-banana-cartoon-workflow-28043a87'
parser=argparse.ArgumentParser();parser.add_argument('--paced',action='store_true');parser.add_argument('--section',type=int);parser.add_argument('--master-section',type=int);parser.add_argument('--threads',type=int,default=8);parser.add_argument('--plan',default='Complete narration repair plan.json');parser.add_argument('--export',default='complete-narration');parser.add_argument('--name',default='Frame 01 - Nano Banana cartoon workflow - complete narration.mp4');parser.add_argument('--label');parser.add_argument('--join-speech',action='store_true');args=parser.parse_args()
assert all(Path(value).name==value for value in[args.plan,args.export,args.name])
plan=json.loads((p/args.plan).read_text())
assert not(args.paced and(args.section is not None or args.master_section is not None))
assert not(args.section is not None and args.master_section is not None)
selected=args.master_section if args.master_section is not None else args.section
section=plan['sections'][selected]if selected is not None else None
media=p/plan['aroll']if args.paced else p/f'exports/{args.export}/section-{args.section:02}.mp4'if args.section is not None else p/'exports'/args.export/args.name
assert media.exists(),media
digest=hashlib.sha256(media.read_bytes()).hexdigest()
out=task/'work/complete-narration-audio'/digest[:16];out.mkdir(parents=True,exist_ok=True)
os.environ.update(HF_HUB_OFFLINE='1',TRANSFORMERS_OFFLINE='1',CUDA_VISIBLE_DEVICES='',HF_HOME=str(out/'hf-cache'))
import torch,soundfile as sf
from transformers import AutoProcessor,AutoModelForMultimodalLM

wav=out/'audio.wav'
if not wav.exists():subprocess.run([r'C:\ffmpeg\bin\ffmpeg.exe','-v','error','-i',str(media),'-vn','-ac','1','-ar','16000',str(wav)],check=True)
audio,sr=sf.read(wav,dtype='float32');assert sr==16000
norm=lambda text:re.findall(r'[a-z0-9]+',text.lower().replace('open art','openart').replace('sea dance','seedance').replace('90s','nineties'))
cues=[]
for block in(p/'Replacement A-roll captions.srt').read_text(encoding='utf-8').strip().split('\n\n'):
    number,timing,*lines=block.splitlines();cues.append({'cue':int(number),'text':' '.join(lines)})
order=list(range(1,178))if args.paced else[1,4,2,3]+list(range(5,178))
assert len(order)==len(set(order))==177
if section:
    anchors=json.loads((p/'Acoustic section anchors.json').read_text())
    boundaries=[0]+[j['cues'][0]for j in anchors['jobs']]+[177]
    order=[i for i in order if boundaries[selected]<i<=boundaries[selected+1]]
expected=norm(' '.join(cues[i-1]['text']for i in order))
ranges=[{'from':0,'to':round(len(audio)/sr*30)}]if args.paced else plan['speech']
if section:
    offset=0 if args.master_section is not None else section['from']
    ranges=[{'from':max(c['from'],section['from'])-offset,'to':min(c['to'],section['to'])-offset}for c in ranges if c['from']<section['to']and c['to']>section['from']]
joined_intervals=[]
if args.join_speech:
    assert not args.paced and section is None
    portions=[];cursor=0
    for cut in ranges:
        portion=audio[round(cut['from']/30*sr):min(len(audio),round(cut['to']/30*sr))]
        portions.append(portion);joined_intervals.append({'masterFrom':cut['from']/30,'masterTo':cut['to']/30,'joinedFrom':cursor/sr,'joinedTo':(cursor+len(portion))/sr});cursor+=len(portion)
    import numpy as np
    audio=np.concatenate(portions)
    ranges=[{'from':0,'to':len(audio)/sr*30}]
requests=[]
for i,cut in enumerate(ranges):
    a=cut['from']/30;b=min(cut['to']/30,len(audio)/sr);n=math.ceil((b-a)/28)
    for j in range(n):
        start=a+j*28;end=min(b,start+29)
        requests.append({'speechInterval':i,'chunk':j,'start':start,'end':end,'key':f'speech-{i:02}-chunk-{j:03}'})
assert 1<=args.threads<=16
torch.set_num_threads(args.threads)
install=Path(r'C:\Users\jonst\Documents\Codex\Local Models\Qwen3-ASR-1.7B-hf/model')
processor=AutoProcessor.from_pretrained(install,local_files_only=True)
model=AutoModelForMultimodalLM.from_pretrained(install,local_files_only=True,dtype=torch.float32,device_map='cpu',attn_implementation='sdpa')
rows=[]
for offset in range(0,len(requests),2):
    batch=requests[offset:offset+2];missing=[r for r in batch if not(out/(r['key']+'.json')).exists()]
    if missing:
        began=time.monotonic();clips=[audio[round(r['start']*sr):round(r['end']*sr)]for r in missing]
        inputs=processor.apply_transcription_request(audio=clips,language='English').to(model.device,model.dtype)
        with torch.inference_mode():generated=model.generate(**inputs,max_new_tokens=256,do_sample=False)
        decoded=processor.decode(generated[:,inputs['input_ids'].shape[1]:],return_format='parsed')
        for request,result in zip(missing,decoded):(out/(request['key']+'.json')).write_text(json.dumps({**request,'batchCpuSeconds':time.monotonic()-began,**result},indent=2),encoding='utf-8')
        del inputs,generated
    for request in batch:rows.append(json.loads((out/(request['key']+'.json')).read_text()))
    print(json.dumps({'completed':len(rows),'total':len(requests),'lastSeconds':rows[-1]['end'],'lastText':rows[-1]['transcription'][-100:]}),flush=True)
heard=[];merges=[]
for row in rows:
    words=norm(row['transcription']);overlap=0
    if row['chunk']:
        for k in range(2,min(12,len(heard),len(words))+1):
            if heard[-k:]==words[:k]:overlap=k
    heard+=words[overlap:];merges.append({'key':row['key'],'overlappingTokensRemoved':overlap})
matcher=difflib.SequenceMatcher(None,expected,heard,autojunk=False)
differences=[{'kind':tag,'expected':' '.join(expected[a:b]),'heard':' '.join(heard[c:d]),'expectedContext':' '.join(expected[max(0,a-8):min(len(expected),b+8)])}for tag,a,b,c,d in matcher.get_opcodes()if tag!='equal']
matched=sum(m.size for m in matcher.get_matching_blocks())
report={'media':str(media),'sha256':digest,'mode':'paced-source'if args.paced else'section'if section else'complete-master','section':args.section,'audioSeconds':len(audio)/sr,'model':'Existing Qwen3-ASR-1.7B, CPU, offline','all177CuesExpected':not bool(section),'expectedCueIds':order,'expectedTokens':len(expected),'heardTokens':len(heard),'matchedTokens':matched,'expectedTokenCoverage':matched/len(expected),'differences':differences,'chunks':rows,'overlapMerges':merges,'limitations':['Independent ASR can misrecognize names and words at chunk boundaries; inspect differences before accepting speech content.','The master pass transcribes actual output narration intervals, excluding demonstration gaps. It does not certify prosody, voice identity or phoneme-to-mouth lip sync.']}
if args.join_speech:
    report['joinedNarrationIntervals']=joined_intervals
    report['chunkTimebase']='Concatenated actual master narration, demo gaps excluded; joinedNarrationIntervals maps to output time.'
dest=task/'outputs'/((args.label+' speech verification.json')if args.label else'Complete paced narration speech verification.json'if args.paced else f'Complete master section-{args.master_section:02} speech verification.json'if args.master_section is not None else f'Complete section-{args.section:02} speech verification.json'if section else'Complete master speech verification.json')
dest.write_text(json.dumps(report,indent=2),encoding='utf-8')
print(json.dumps({k:v for k,v in report.items()if k not in['differences','chunks','overlapMerges','limitations']}),flush=True)
